function getDateKey(date = new Date()) {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

function getCalendarDayDifference(fromDate: string, toDate: string) {
  const [fromYear, fromMonth, fromDay] = fromDate.split('-').map(Number);
  const [toYear, toMonth, toDay] = toDate.split('-').map(Number);
  const fromUtc = Date.UTC(fromYear, fromMonth - 1, fromDay);
  const toUtc = Date.UTC(toYear, toMonth - 1, toDay);
  return Math.round((toUtc - fromUtc) / (24 * 60 * 60 * 1000));
}

function ProgressDots({ active }: { active: number }) {
  return (
    <View style={styles.dotsRow}>
      {[0, 1, 2].map((i) => (
        <View
          key={i}
          style={[styles.dot, { backgroundColor: i <= active ? colors.gold : '#E3DAC5' }]}
        />
      ))}
    </View>
  );
}

function JournalProgressDots({
  currentIndex,
  allComplete,
}: {
  currentIndex: number;
  allComplete: boolean;
}) {
  return (
    <View style={styles.journalDots}>
      {LESSONS.map((lesson, index) => (
        <View
          key={lesson.id}
          style={[
            styles.journalDot,
            (index < currentIndex || allComplete) && styles.journalDotComplete,
            index === currentIndex && !allComplete && styles.journalDotCurrent,
            index > currentIndex && styles.journalDotUpcoming,
          ]}
        />
      ))}
    </View>
  );
}

function Quiz({
  lesson,
  selected,
  setSelected,
  onContinue,
  scenario = false,
}: {
  lesson: Lesson;
  selected: string | null;
  setSelected: (value: string | null) => void;
  onContinue: () => void;
  scenario?: boolean;
}) {
  const answered = selected !== null;
  const options = scenario ? lesson.scenario.options : lesson.options;
  const question = scenario ? lesson.scenario.question : lesson.question;
  const selectedOption = options.find((o) => o.key === selected);

  return (
    <View style={styles.screen}>
      <ProgressDots active={1} />
      <Text style={styles.eyebrow}>{scenario ? 'Scenario' : 'Quiz'} · Concept {lesson.id}</Text>
      <Text style={styles.title}>{question}</Text>
      <View style={{ marginTop: 8 }}>
        {options.map((o) => {
          const isSelected = selected === o.key;
          const showCorrect = answered && o.correct;
          const showWrong = answered && isSelected && !o.correct;
          return (
            <TouchableOpacity
              key={o.key}
              disabled={answered}
              onPress={() => setSelected(o.key)}
              style={[
                styles.option,
                showCorrect && styles.optionCorrect,
                showWrong && styles.optionWrong,
              ]}
            >
              <View
                style={[
                  styles.optionBadge,
                  { backgroundColor: showCorrect ? colors.green : showWrong ? colors.red : colors.navy },
                ]}
              >
                <Text style={styles.optionBadgeText}>{o.key}</Text>
              </View>
              <Text style={styles.optionText}>{o.text}</Text>
            </TouchableOpacity>
          );
        })}
      </View>
      {answered && selectedOption && (
        <Text style={styles.feedback}>
          {selectedOption.correct ? lesson.correctFeedback : lesson.incorrectFeedback}
        </Text>
      )}
      {answered && (
        <TouchableOpacity style={[styles.button, styles.bottomButton]} onPress={onContinue}>
          <Text style={styles.buttonText}>Continue</Text>
        </TouchableOpacity>
      )}
    </View>
  );
}

export default function App() {
  const [fontsLoaded] = useFonts({
    Inter_400Regular,
    Inter_500Medium,
    Inter_600SemiBold,
    Inter_700Bold,
    Fraunces_700Bold,
  });
  const [screen, setScreen] = useState<'home' | 'learn' | 'quiz' | 'scenario' | 'result'>('home');
  const [currentLessonIndex, setCurrentLessonIndex] = useState(0);
  const [selected, setSelected] = useState<string | null>(null);
  const [xp, setXp] = useState(0);
  const [streak, setStreak] = useState(3);
  const [lastCompletionDate, setLastCompletionDate] = useState<string | null>(null);
  const [earnedXp, setEarnedXp] = useState(0);
  const [allLessonsComplete, setAllLessonsComplete] = useState(false);
  const [currentScenarioIndex, setCurrentScenarioIndex] = useState(0);
  const [allScenariosComplete, setAllScenariosComplete] = useState(false);
  const [hasLoadedProgress, setHasLoadedProgress] = useState(false);
  const [notificationsReady, setNotificationsReady] = useState(false);
  const notificationId = useRef<string | null>(null);

  useEffect(() => {
    let isMounted = true;

    AsyncStorage.getItem(STORAGE_KEY)
      .then((stored) => {
        if (!isMounted || !stored) {
          setHasLoadedProgress(true);
          return;
        }

        try {
          const parsed = JSON.parse(stored) as {
            xp?: number;
            streak?: number;
            lastCompletionDate?: string;
            currentLessonIndex?: number;
            allLessonsComplete?: boolean;
            currentScenarioIndex?: number;
            allScenariosComplete?: boolean;
          };
          if (typeof parsed.xp === 'number') {
            setXp(parsed.xp);
          }
          if (typeof parsed.streak === 'number') {
            setStreak(parsed.streak);
          }
          if (typeof parsed.lastCompletionDate === 'string') {
            setLastCompletionDate(parsed.lastCompletionDate);
          }
          if (
            typeof parsed.currentLessonIndex === 'number' &&
            parsed.currentLessonIndex >= 0 &&
            parsed.currentLessonIndex < LESSONS.length
          ) {
            setCurrentLessonIndex(parsed.currentLessonIndex);
          }
          if (typeof parsed.allLessonsComplete === 'boolean') {
            setAllLessonsComplete(parsed.allLessonsComplete);
          }
          if (
            typeof parsed.currentScenarioIndex === 'number' &&
            parsed.currentScenarioIndex >= 0 &&
            parsed.currentScenarioIndex < LESSONS.length
          ) {
            setCurrentScenarioIndex(parsed.currentScenarioIndex);
          }
          if (typeof parsed.allScenariosComplete === 'boolean') {
            setAllScenariosComplete(parsed.allScenariosComplete);
          }
        } catch {
          // Ignore malformed data and keep defaults.
        } finally {
          setHasLoadedProgress(true);
        }
      })
      .catch(() => {
        if (isMounted) {
          setHasLoadedProgress(true);
        }
      });

    return () => {
      isMounted = false;
    };
  }, []);

  useEffect(() => {
    if (!hasLoadedProgress) {
      return;
    }

    AsyncStorage.setItem(
          STORAGE_KEY,
          JSON.stringify({
            xp,
            streak,
            lastCompletionDate,
            currentLessonIndex,
            allLessonsComplete,
            currentScenarioIndex,
            allScenariosComplete,
          }),
    ).catch(() => {
      // Ignore storage write errors in this demo app.
    });
  }, [
    xp,
    streak,
    lastCompletionDate,
    currentLessonIndex,
    allLessonsComplete,
    currentScenarioIndex,
    allScenariosComplete,
    hasLoadedProgress,
  ]);

  useEffect(() => {
    if (Platform.OS === 'web') {
      return;
    }

    let isMounted = true;

    Notifications.requestPermissionsAsync()
      .then(({ status }) => {
        if (isMounted && status === Notifications.PermissionStatus.GRANTED) {
          setNotificationsReady(true);
        }
      })
      .catch(() => {
        // Ignore denied or unavailable notification permissions.
      });

    return () => {
      isMounted = false;
    };
  }, []);

  useEffect(() => {
    if (Platform.OS === 'web' || !notificationsReady || !hasLoadedProgress) {
      return;
    }

    let isMounted = true;
    const scheduleReminder = async () => {
      await Notifications.cancelAllScheduledNotificationsAsync();
      notificationId.current = null;

      const lesson = LESSONS[currentLessonIndex];
      const body = allLessonsComplete ? 'Come back to Folient!' : lesson?.teaser ?? 'Come back to Folient!';
      const id = await Notifications.scheduleNotificationAsync({
        content: { title: 'Folient', body },
        trigger: {
          type: Notifications.SchedulableTriggerInputTypes.DAILY,
          hour: 19,
          minute: 0,
        },
      });

      if (isMounted) {
        notificationId.current = id;
      } else {
        await Notifications.cancelScheduledNotificationAsync(id);
      }
    };

    scheduleReminder().catch(() => {
      // Ignore scheduling errors so the lesson flow keeps working.
    });

    return () => {
      isMounted = false;
    };
  }, [currentLessonIndex, allLessonsComplete, hasLoadedProgress, notificationsReady]);

  if (!fontsLoaded) return <View style={styles.container} />;

  const finishLearn = () => {
    setXp((currentXp) => currentXp + 5);
    setEarnedXp(5);
    setScreen('quiz');
  };

  const completeLesson = (selectedOptionKey: string | null) => {
    const lesson = LESSONS[currentLessonIndex];
    const today = getDateKey();
    const daysSinceLastCompletion = lastCompletionDate
      ? getCalendarDayDifference(lastCompletionDate, today)
      : null;
    const nextStreak =
      daysSinceLastCompletion === 0
        ? streak
        : daysSinceLastCompletion === 1 || daysSinceLastCompletion === 2
          ? streak + 1
          : 1;
    const quizXp = lesson.options.find((option) => option.key === selectedOptionKey)?.correct ? 10 : 0;

    setXp((currentXp) => currentXp + quizXp);
    setEarnedXp((currentEarnedXp) => currentEarnedXp + quizXp);
    setStreak(nextStreak);
    setLastCompletionDate(today);
    setScreen('result');
  };

  const completeScenario = (selectedOptionKey: string | null) => {
    const scenario = LESSONS[currentScenarioIndex].scenario;
    const today = getDateKey();
    const daysSinceLastCompletion = lastCompletionDate
      ? getCalendarDayDifference(lastCompletionDate, today)
      : null;
    const nextStreak =
      daysSinceLastCompletion === 0
        ? streak
        : daysSinceLastCompletion === 1 || daysSinceLastCompletion === 2
          ? streak + 1
          : 1;
    const scenarioXp = scenario.options.find((option) => option.key === selectedOptionKey)?.correct ? 15 : 0;

    setXp((currentXp) => currentXp + scenarioXp);
    setEarnedXp(scenarioXp);
    setStreak(nextStreak);
    setLastCompletionDate(today);
    setScreen('result');
  };

  const continueAfterResult = () => {
    if (allLessonsComplete) {
      if (currentScenarioIndex < LESSONS.length - 1) {
        setCurrentScenarioIndex((currentIndex) => currentIndex + 1);
      } else {
        setAllScenariosComplete(true);
      }
    } else {
      if (currentLessonIndex < LESSONS.length - 1) {
        setCurrentLessonIndex((currentIndex) => currentIndex + 1);
      } else {
        setAllLessonsComplete(true);
      }
    }
    setScreen('home');
    setSelected(null);
    setEarnedXp(0);
  };

  return (
    <SafeAreaView style={styles.container}>
      {screen === 'home' && (
        <Home
          lesson={LESSONS[currentLessonIndex]}
          allLessonsComplete={allLessonsComplete}
          allScenariosComplete={allScenariosComplete}
          currentLessonIndex={currentLessonIndex}
          currentScenarioIndex={currentScenarioIndex}
          onStart={() => setScreen('learn')}
          onStartScenario={() => setScreen('scenario')}
          xp={xp}
          streak={streak}
        />
      )}
      {screen === 'learn' && <Learn lesson={LESSONS[currentLessonIndex]} onContinue={finishLearn} />}
      {screen === 'quiz' && (
        <Quiz
          lesson={LESSONS[currentLessonIndex]}
          selected={selected}
          setSelected={setSelected}
          onContinue={() => completeLesson(selected)}
        />
      )}
      {screen === 'scenario' && (
        <Quiz
          lesson={LESSONS[currentScenarioIndex]}
          selected={selected}
          setSelected={setSelected}
          onContinue={() => completeScenario(selected)}
          scenario
        />
      )}
      {screen === 'result' && (
        <Result
          lesson={LESSONS[currentLessonIndex]}
          onDone={continueAfterResult}
          streak={streak}
          earnedXp={earnedXp}
        />
      )}
    </SafeAreaView>
  );
}
